3 월별 이행률 자동 계산 + 코스 저장 로직

미션 완료 시 이행률 누적 · 코스 자동 결정

기존 앱의 미션 완료 버튼에 연결하는 이행률 추적 엔진입니다.
localStorage에 월별 데이터를 저장하며, 달이 바뀌면 자동으로 코스를 재계산합니다.

📌 연결 위치: 기존 미션 완료 버튼의 onclick 핸들러 안에 recordMissionComplete() 호출 추가.
📌 성찰 일기 저장: 기존 "저장하기" 버튼에 recordReflectionSaved() 호출 추가.
📌 마음챙김 완료: 기존 타이머 완료 시 recordMindfulnessDone() 호출 추가.

▼ JavaScript — <script> 태그 안에 추가

/* ── [기능3] 월별 이행률 자동 계산 + 코스 저장 로직 ── */ // localStorage 키 상수 const STORAGE_KEY = 'mc_monthly_data'; // 이번 달 키 (예: "2025-06") function getMonthKey() { const d = new Date(); return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`; } // 이번 달 데이터 불러오기 (없으면 초기화) function getMonthlyData() { const key = getMonthKey(); const stored = localStorage.getItem(STORAGE_KEY); const all = stored ? JSON.parse(stored) : {}; if (!all[key]) { all[key] = { totalMissions: 0, // 이번 달 목표 미션 수 (주 5일 × 5주 = 25 기본) completedMissions: 0, reflectionsSaved: 0, mindfulnessDone: 0, courseLevel: all[getPrevMonthKey()] ? getNextCourse(calcCompletionRate(all[getPrevMonthKey()])) : 'blue', // 첫 달은 표준 코스 시작 startDate: new Date().toISOString() }; localStorage.setItem(STORAGE_KEY, JSON.stringify(all)); } return { key, all, data: all[key] }; } // 이전 달 키 function getPrevMonthKey() { const d = new Date(); d.setMonth(d.getMonth() - 1); return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`; } // 이행률 계산 function calcCompletionRate(data) { if (!data || data.totalMissions === 0) return 0; // 미션 완료 60% + 성찰 일기 20% + 마음챙김 20% 가중치 합산 const missionScore = Math.min(data.completedMissions / data.totalMissions, 1) * 60; const reflectionScore = Math.min(data.reflectionsSaved / 20, 1) * 20; const mindfulScore = Math.min(data.mindfulnessDone / 10, 1) * 20; return Math.round(missionScore + reflectionScore + mindfulScore); } // ── 각 활동 완료 시 호출하는 함수들 ── function recordMissionComplete() { const { key, all, data } = getMonthlyData(); data.completedMissions += 1; all[key] = data; localStorage.setItem(STORAGE_KEY, JSON.stringify(all)); updateJourneyUI(); // 기능1 UI 갱신 } function recordReflectionSaved() { const { key, all, data } = getMonthlyData(); data.reflectionsSaved += 1; all[key] = data; localStorage.setItem(STORAGE_KEY, JSON.stringify(all)); updateJourneyUI(); } function recordMindfulnessDone() { const { key, all, data } = getMonthlyData(); data.mindfulnessDone += 1; all[key] = data; localStorage.setItem(STORAGE_KEY, JSON.stringify(all)); updateJourneyUI(); } // 목표 미션 수 설정 (월 시작 시 호출) function setMonthlyMissionTarget(count) { const { key, all, data } = getMonthlyData(); data.totalMissions = count; all[key] = data; localStorage.setItem(STORAGE_KEY, JSON.stringify(all)); } // 기능1 여정 카드 UI 실시간 갱신 function updateJourneyUI() { const { data } = getMonthlyData(); const rate = calcCompletionRate(data); JOURNEY_DATA.completionRate = rate; renderJourneyCard('journey-card', JOURNEY_DATA); } // 현재 이행률 조회 (외부에서 사용) function getCurrentCompletionRate() { const { data } = getMonthlyData(); return calcCompletionRate(data); } // 현재 코스 레벨 조회 function getCurrentCourseLevel() { const { data } = getMonthlyData(); return data.courseLevel; } // ── 기존 미션 완료 버튼에 연결 예시 ── // 기존 코드: // completeMission() { ... addXP(40); showPopup(); } // // 수정 후: // completeMission() { ... addXP(40); showPopup(); recordMissionComplete(); } // // 기존 성찰 일기 저장 버튼: // saveReflection() { ... addXP(20); } // 수정 후: // saveReflection() { ... addXP(20); recordReflectionSaved(); } // // 기존 마음챙김 타이머 완료: // onTimerComplete() { ... addXP(30); } // 수정 후: // onTimerComplete() { ... addXP(30); recordMindfulnessDone(); }

⚠️ 주의사항

• 위 코드의 < > HTML 엔티티는 코드블록 디스플레이용입니다. 실제 파일에 붙여넣을 때는 정상 태그 문자(< >)로 사용하세요.

• CSS 변수(--card, --purple 등)는 기존 앱의 :root에 이미 유사한 값이 있다면 변수명을 맞춰 수정하세요.

• getNextCourse() 함수는 기능1과 기능2에서 공통 사용 — 파일에 한 번만 선언하세요.
